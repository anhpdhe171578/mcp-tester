# GitHub Push Commands for MCP Tester
# Thay YOUR_USERNAME bằng username GitHub của bạn

# 1. Thêm remote repository
git remote add origin https://github.com/anhpdhe171578/mcp-tester.git

# 2. Đổi tên branch main (nếu cần)
git branch -M main

# 3. Push code lên GitHub
git push -u origin main

# Hoặc nếu muốn giữ tên master:
# git push -u origin master
